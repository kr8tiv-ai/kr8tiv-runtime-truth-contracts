```typescript
import { writable, derived } from 'svelte/store';
import type { WizardState, WizardScreen, WIZARD_SCREENS } from '../types';

const initialState: WizardState = {
  currentScreen: 'welcome',
  walletAddress: null,
  walletType: null,
  nftVerified: false,
  companionId: null,
  companionName: null,
  hardware: null,
  selectedQuant: null,
  downloadProgress: 0,
  downloadSpeed: '',
  error: null,
};

export const wizardState = writable<WizardState>(initialState);

export const currentScreenIndex = derived(wizardState, ($state) =>
  WIZARD_SCREENS.indexOf($state.currentScreen)
);

export function goToScreen(screen: WizardScreen) {
  wizardState.update(s => ({ ...s, currentScreen: screen, error: null }));
}

export function nextScreen() {
  wizardState.update(s => {
    const idx = WIZARD_SCREENS.indexOf(s.currentScreen);
    if (idx < WIZARD_SCREENS.length - 1) {
      return { ...s, currentScreen: WIZARD_SCREENS[idx + 1], error: null };
    }
    return s;
  });
}

export function prevScreen() {
  wizardState.update(s => {
    const idx = WIZARD_SCREENS.indexOf(s.currentScreen);
    if (idx > 0) {
      return { ...s, currentScreen: WIZARD_SCREENS[idx - 1], error: null };
    }
    return s;
  });
}

export function setError(error: string) {
  wizardState.update(s => ({ ...s, error }));
}

export function resetWizard() {
  wizardState.set(initialState);
}
```

```typescript
// Wallet connection for dual UX: Phantom (Solana) and MetaMask (EVM)
// In Tauri desktop context, wallets communicate via deeplinks or injected providers

export interface WalletConnection {
  address: string;
  type: 'phantom' | 'metamask';
  signMessage: (message: string) => Promise<string>;
}

export async function connectPhantom(): Promise<WalletConnection> {
  // Phantom desktop: check for window.phantom?.solana (if webview has extension)
  // Fallback: use Phantom deeplink protocol for desktop
  // For MVP: Simulate connection with wallet adapter pattern
  // Real implementation will use @solana/wallet-adapter-phantom
  
  const phantom = (window as any)?.phantom?.solana;
  if (!phantom) {
    throw new Error('Phantom wallet not detected. Install Phantom browser extension or desktop app.');
  }
  
  const response = await phantom.connect();
  const address = response.publicKey.toString();
  
  return {
    address,
    type: 'phantom',
    signMessage: async (message: string) => {
      const encodedMessage = new TextEncoder().encode(message);
      const signature = await phantom.signMessage(encodedMessage, 'utf8');
      return Buffer.from(signature.signature).toString('base64');
    },
  };
}

export async function connectMetaMask(): Promise<WalletConnection> {
  // MetaMask: check for window.ethereum (EIP-1193 provider)
  const ethereum = (window as any)?.ethereum;
  if (!ethereum || !ethereum.isMetaMask) {
    throw new Error('MetaMask not detected. Install MetaMask browser extension or desktop app.');
  }
  
  const accounts = await ethereum.request({ method: 'eth_requestAccounts' });
  const address = accounts[0];
  
  return {
    address,
    type: 'metamask',
    signMessage: async (message: string) => {
      const signature = await ethereum.request({
        method: 'personal_sign',
        params: [message, address],
      });
      return signature as string;
    },
  };
}

export function truncateAddress(address: string): string {
  if (address.length <= 12) return address;
  return `${address.slice(0, 6)}...${address.slice(-4)}`;
}
```

```typescript
import type { NftVerification } from './types';
import type { WalletConnection } from './wallet';

// NFT verification: check on-chain ownership, gate per companion
// For Solana: check Metaplex NFT ownership via RPC
// For EVM: check ERC-721 ownership via contract call

const KR8TIV_API = 'https://api.kr8tiv.ai';  // Backend verification endpoint
const VERIFICATION_MESSAGE = 'KIN Companion Verification: Sign to prove NFT ownership';

export async function verifyNftOwnership(wallet: WalletConnection): Promise<NftVerification> {
  // Step 1: Sign verification message
  const signature = await wallet.signMessage(VERIFICATION_MESSAGE);
  
  // Step 2: Send to kr8tiv backend for on-chain verification
  // Backend checks: wallet owns a KIN Genesis NFT, returns companion details
  const response = await fetch(`${KR8TIV_API}/verify-nft`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({
      walletAddress: wallet.address,
      walletType: wallet.type,
      signature,
      message: VERIFICATION_MESSAGE,
    }),
  });
  
  if (!response.ok) {
    const error = await response.json().catch(() => ({ message: 'Verification failed' }));
    throw new Error(error.message || 'NFT verification failed. Make sure you hold a KIN Genesis NFT.');
  }
  
  const data: NftVerification = await response.json();
  return data;
}

export async function checkOfflineGrace(): Promise<NftVerification | null> {
  // Check for cached verification (30-day grace period for offline use)
  // Stored in Tauri app data directory
  try {
    const { invoke } = await import('@tauri-apps/api/core');
    const cached = await invoke<string>('read_cached_verification');
    if (!cached) return null;
    
    const verification: NftVerification & { cachedAt: number } = JSON.parse(cached);
    const thirtyDays = 30 * 24 * 60 * 60 * 1000;
    if (Date.now() - verification.cachedAt > thirtyDays) return null;
    
    return verification;
  } catch {
    return null;
  }
}
```

```rust
// Model download manager with progress events and resume support
// Uses Ollama's pull API for model downloads, or direct HTTP for GGUF files

use tauri::{AppHandle, Emitter};
use serde::Serialize;

#[derive(Clone, Serialize)]
struct DownloadProgress {
    percent: f64,        // 0.0 - 100.0
    downloaded_mb: f64,
    total_mb: f64,
    speed_mbps: f64,     // MB/s
    status: String,      // "downloading", "verifying", "complete", "error"
}

#[tauri::command]
pub async fn download_model(
    app: AppHandle,
    model_name: String,   // e.g., "kr8tiv/cipher:q4_k_m"
) -> Result<String, String> {
    // Strategy: Use Ollama's /api/pull endpoint which handles:
    // - Resumable downloads (layer-based, skips existing layers)
    // - Integrity verification (SHA256 per layer)
    // - Progress reporting via streaming JSON
    
    let client = reqwest::Client::new();
    let response = client
        .post("http://localhost:11434/api/pull")
        .json(&serde_json::json!({ "name": model_name, "stream": true }))
        .send()
        .await
        .map_err(|e| format!("Failed to connect to Ollama: {}. Is Ollama running?", e))?;
    
    if !response.status().is_success() {
        return Err(format!("Ollama returned error: {}", response.status()));
    }
    
    // Stream progress events to frontend
    let mut stream = response.bytes_stream();
    let mut last_emit = std::time::Instant::now();
    let start_time = std::time::Instant::now();
    let mut total_downloaded: f64 = 0.0;
    
    use futures_util::StreamExt;
    while let Some(chunk) = stream.next().await {
        let chunk = chunk.map_err(|e| format!("Download stream error: {}", e))?;
        let text = String::from_utf8_lossy(&chunk);
        
        // Parse Ollama's streaming JSON (one JSON object per line)
        for line in text.lines() {
            if let Ok(json) = serde_json::from_str::<serde_json::Value>(line) {
                let status = json["status"].as_str().unwrap_or("unknown");
                let completed = json["completed"].as_f64().unwrap_or(0.0);
                let total = json["total"].as_f64().unwrap_or(1.0);
                
                total_downloaded = completed;
                let elapsed = start_time.elapsed().as_secs_f64();
                let speed = if elapsed > 0.0 { (completed / 1_048_576.0) / elapsed } else { 0.0 };
                let percent = if total > 0.0 { (completed / total) * 100.0 } else { 0.0 };
                
                // Emit progress at most every 250ms to avoid flooding
                if last_emit.elapsed().as_millis() >= 250 || status == "success" {
                    let progress = DownloadProgress {
                        percent,
                        downloaded_mb: completed / 1_048_576.0,
                        total_mb: total / 1_048_576.0,
                        speed_mbps: speed,
                        status: if status == "success" { "complete".to_string() } else { "downloading".to_string() },
                    };
                    let _ = app.emit("download-progress", &progress);
                    last_emit = std::time::Instant::now();
                }
            }
        }
    }
    
    Ok("Model downloaded successfully".to_string())
}

#[tauri::command]
pub async fn cancel_download() -> Result<String, String> {
    // Ollama handles cleanup -- killing the connection aborts the pull
    // The next pull will resume from where it left off (layer-based resume)
    Ok("Download cancelled. Progress is saved and will resume next time.".to_string())
}
```
