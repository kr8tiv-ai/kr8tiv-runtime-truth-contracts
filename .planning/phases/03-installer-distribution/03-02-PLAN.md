// --- Hardware Detection ---

export type GpuVendor = 'nvidia' | 'amd' | 'intel' | 'apple' | 'none';

export type GpuTier = 'high' | 'medium' | 'low' | 'cpu_only';

export interface GpuInfo {
  vendor: GpuVendor;
  model: string;            // e.g., "NVIDIA RTX 4090", "Apple M2 Pro"
  vram_mb: number;          // VRAM in megabytes, 0 for integrated/CPU-only
  driver_version: string;   // e.g., "535.129.03" for NVIDIA
  compute_capability: string; // e.g., "8.9" for NVIDIA, empty for others
  metal_supported: boolean; // macOS Metal support
}

export interface HardwareProfile {
  gpu: GpuInfo;
  gpu_tier: GpuTier;
  ram_gb: number;            // System RAM in GB (rounded)
  disk_free_gb: number;      // Available disk space in GB
  os: 'windows' | 'macos' | 'linux';
  cpu_cores: number;
  recommended_quant: QuantLevel;
  can_run_gpu: boolean;      // false = CPU-only inference
  warnings: string[];        // Human-readable warnings for the wizard
}

// --- Model / Quantization ---

export type QuantLevel = 'Q8_0' | 'Q5_K_M' | 'Q4_K_M' | 'Q3_K_M';

  ram_required_gb: number;

Build the hardware detection backend in Rust and a TypeScript frontend wrapper.

**installer/src-tauri/src/hardware.rs:**

Add to Cargo.toml dependencies: `sysinfo = "0.32"`, `wgpu = { version = "24", features = ["vulkan-portability"] }` (for cross-platform GPU detection).

Implement:

1. `#[tauri::command] pub async fn detect_hardware() -> Result<HardwareProfile, String>` that returns a serializable struct matching the TypeScript HardwareProfile type.

2. **GPU detection strategy (in order of preference):**
   - **Windows NVIDIA:** Run `nvidia-smi --query-gpu=name,memory.total,driver_version,compute_cap --format=csv,noheader,nounits` via `std::process::Command`. Parse output. If nvidia-smi not found or fails, GPU is not NVIDIA.
   - **Windows AMD:** Check for `C:\Windows\System32\DriverStore\FileRepository\*amd*` or run `wmic path win32_VideoController get Name,AdapterRAM`. Note: AMD GPUs on Windows have NO ROCm support -- set warning "AMD GPU detected but GPU acceleration not available on Windows. Inference will use CPU."
   - **macOS:** Use `system_profiler SPDisplaysDataType -json` to get GPU info. Check for "Metal" support. Apple Silicon always has Metal. Set `metal_supported: true` for M-series chips.
   - **Fallback:** Use `sysinfo` crate to get basic GPU info from system. If all detection fails, return `vendor: "none"`, `can_run_gpu: false`.

3. **GPU tier classification:**
   - `high`: VRAM >= 16384 MB (16GB+) OR Apple Silicon with 32GB+ unified memory
   - `medium`: VRAM >= 8192 MB (8-15GB) OR Apple Silicon with 16-31GB
   - `low`: VRAM >= 4096 MB (4-7GB) OR Apple Silicon with 8-15GB
   - `cpu_only`: No discrete GPU or VRAM < 4096

4. **Quantization recommendation algorithm:**
   ```
   if gpu_tier == high AND ram_gb >= 36:  Q8_0
   elif gpu_tier >= medium AND ram_gb >= 32: Q5_K_M
   elif ram_gb >= 22: Q4_K_M   (the sweet spot, default for most users)
   elif ram_gb >= 18: Q3_K_M   (with warning about quality loss)
   else: Q4_K_M with CPU-only warning (better quality at slow speed > fast garbage)
   ```
   Per research: NEVER default to Q3_K_M. The perplexity penalty (+0.24 vs +0.05) kills companion personality. If RAM is insufficient for Q4_K_M, warn the user but still recommend Q4_K_M with CPU inference rather than Q3_K_M on GPU.

5. **Warning generation:**
   - NVIDIA driver < 531: "Your NVIDIA driver is outdated. GPU acceleration may fail. Update at nvidia.com/drivers"
   - AMD on Windows: "AMD GPUs don't support GPU acceleration on Windows. Your companion will run on CPU (slower but works)."
   - RAM < 16GB: "Your system has {N}GB RAM. The companion may be slow. 16GB+ recommended."
   - Disk free < 25GB: "Low disk space ({N}GB free). The companion model needs at least 20GB."
   - CPU-only detected: "No compatible GPU detected. Your companion will run on CPU. Expect slower responses."

6. **RAM and disk detection:** Use `sysinfo` crate. `System::new_all()`, then `sys.total_memory()` for RAM, and `sys.disks()` for free space on the install drive.

7. Define Rust structs with `#[derive(Serialize)]` matching the TypeScript types exactly (snake_case in Rust, Tauri auto-converts to camelCase for JS).

**Also update installer/src-tauri/src/main.rs** to add `mod hardware;` and register the `hardware::detect_hardware` command in the invoke_handler.

**installer/src/lib/hardware.ts:**
```typescript
import { invoke } from '@tauri-apps/api/core';
import type { HardwareProfile, QuantLevel, QUANT_OPTIONS } from './types';

export async function detectHardware(): Promise<HardwareProfile> {
  return await invoke<HardwareProfile>('detect_hardware');
}

export function getAvailableQuants(profile: HardwareProfile): typeof QUANT_OPTIONS {
  // Filter QUANT_OPTIONS to only those the user's hardware can run
  // Return all options but mark which ones fit in RAM
  return QUANT_OPTIONS.filter(q => q.ram_required_gb <= profile.ram_gb + 4);
  // +4GB buffer: OS and Ollama overhead, but user might close other apps
}

export function formatVram(mb: number): string {
  if (mb >= 1024) return `${(mb / 1024).toFixed(1)} GB`;
  return `${mb} MB`;
}

export function formatRam(gb: number): string {
  return `${gb} GB`;
}

export function getGpuStatusColor(profile: HardwareProfile): 'green' | 'yellow' | 'red' {
  if (profile.can_run_gpu && profile.gpu_tier === 'high') return 'green';
  if (profile.can_run_gpu) return 'yellow';
  return 'red';
}
```
