**installer/src/lib/services/memory-extractor.ts:**

The extractor uses the local Ollama model (same model ID as the companion) with a non-streaming `/api/generate` call to extract structured memories. This keeps the extraction local and private.

```typescript
import { memoryService } from './memory-db';
import type { MemoryEntry, MemoryCategory, MemoryExtractionResult } from '../types/memory';

const OLLAMA_BASE = 'http://localhost:11434';

// Extraction prompt template -- designed for Gemma4/small models
const EXTRACTION_PROMPT = `Analyze this conversation exchange and extract memorable facts.
Return a JSON array of memory objects. Each object must have:
- content: a single clear sentence describing what to remember (max 100 chars)  
- category: one of "fact" | "preference" | "skill" | "relationship" | "goal"
- importance: 0.0 to 1.0 (0.8+ for explicit preferences, 0.5-0.7 for inferred facts, 0.3-0.5 for casual mentions)

Only extract genuinely memorable information about the USER (not general knowledge).
Return [] if nothing memorable is present.
Return ONLY valid JSON, no markdown, no explanation.

Examples of memorable:
- User says "I prefer dark themes" -> {content: "User prefers dark themes in UI", category: "preference", importance: 0.8}
- User mentions "my DeFi project" -> {content: "User is building a DeFi project", category: "goal", importance: 0.6}
- User demonstrates Three.js knowledge -> {content: "User has Three.js expertise", category: "skill", importance: 0.5}

Examples of NOT memorable:
- General questions about programming concepts
- Requests for help with common tasks
- Greetings and small talk

Conversation:
USER: {userMessage}
ASSISTANT: {assistantMessage}

JSON array:`;

interface ExtractionCandidate {
  content: string;
  category: MemoryCategory;
  importance: number;
}

async function callExtractionModel(modelId: string, prompt: string): Promise<ExtractionCandidate[]> {
  try {
    const response = await fetch(`${OLLAMA_BASE}/api/generate`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        model: modelId,
        prompt,
        stream: false,
        options: {
          temperature: 0.1,    // Low temperature for structured extraction
          top_p: 0.9,
          num_ctx: 2048,       // Extraction prompt is short
        },
        format: 'json',        // Ollama structured output mode
      }),
      signal: AbortSignal.timeout(15000), // 15 second timeout for extraction
    });

    if (!response.ok) return [];

    const data = await response.json() as { response: string };
    const parsed = JSON.parse(data.response);

    // Validate array structure
    if (!Array.isArray(parsed)) return [];

    return parsed.filter((item: unknown): item is ExtractionCandidate => {
      if (typeof item !== 'object' || item === null) return false;
      const obj = item as Record<string, unknown>;
      return (
        typeof obj.content === 'string' &&
        obj.content.length > 0 &&
        obj.content.length <= 200 &&
        ['fact', 'preference', 'skill', 'relationship', 'goal'].includes(obj.category as string) &&
        typeof obj.importance === 'number' &&
        (obj.importance as number) >= 0 &&
        (obj.importance as number) <= 1
      );
    });
  } catch {
    // Extraction is best-effort -- never block the main chat flow
    return [];
  }
}

async function isDuplicate(content: string): Promise<boolean> {
  // FTS search for near-identical content -- if match score is high, skip
  const existing = await memoryService.searchFts(
    // Extract key nouns for FTS query (naive: use first 5 words)
    content.split(' ').slice(0, 5).join(' '),
    3
  );
  return existing.some(m => {
    // Simple similarity: if 70% of words overlap, treat as duplicate
    const existingWords = new Set(m.content.toLowerCase().split(/\s+/));
    const newWords = content.toLowerCase().split(/\s+/);
    const overlap = newWords.filter(w => existingWords.has(w)).length;
    return overlap / newWords.length > 0.7;
  });
}

export async function extractMemories(params: {
  userMessage: string;
  assistantMessage: string;
  sourceMessageId: string;
  conversationId: string;
  modelId: string;          // Companion's model ID for extraction call
}): Promise<MemoryExtractionResult> {
  const result: MemoryExtractionResult = {
    extracted: [],
    updated: [],
    skipped: 0,
  };

  const prompt = EXTRACTION_PROMPT
    .replace('{userMessage}', params.userMessage.slice(0, 500))       // Cap input length
    .replace('{assistantMessage}', params.assistantMessage.slice(0, 500));

  const candidates = await callExtractionModel(params.modelId, prompt);

  for (const candidate of candidates) {
    const duplicate = await isDuplicate(candidate.content);
    if (duplicate) {
      result.skipped++;
      continue;
    }

    const entry = await memoryService.store({
      content: candidate.content,
      category: candidate.category,
      importance: candidate.importance,
      sourceMessageId: params.sourceMessageId,
      conversationId: params.conversationId,
      localOnly: false,
    });

    result.extracted.push(entry);
  }

  return result;
}

export class MemoryExtractor {
  private modelId: string;

  constructor(modelId: string) {
    this.modelId = modelId;
  }

  async extract(params: {
    userMessage: string;
    assistantMessage: string;
    sourceMessageId: string;
    conversationId: string;
  }): Promise<MemoryExtractionResult> {
    return extractMemories({ ...params, modelId: this.modelId });
  }
}
```

**installer/src/lib/services/memory-context.ts:**

```typescript
import { memoryService } from './memory-db';
import type { MemoryEntry, MemoryCategory } from '../types/memory';

// Rough token estimate: 1 token ≈ 4 chars (conservative for English text)
function estimateTokens(text: string): number {
  return Math.ceil(text.length / 4);
}

// Select memories relevant to the current query using FTS + recency + importance
async function selectRelevantMemories(params: {
  query: string;
  limit: number;
}): Promise<MemoryEntry[]> {
  const [ftsResults, recentResults] = await Promise.all([
    // FTS: finds memories semantically related to the current message
    memoryService.searchFts(
      // Use first 8 words of query as FTS terms (avoid stop words bleeding into long queries)
      params.query.split(/\s+/).slice(0, 8).join(' '),
      Math.ceil(params.limit * 0.6)  // 60% from FTS relevance
    ),
    // Recent + high-importance: ensures frequently accessed memories always appear
    memoryService.getRecent(Math.ceil(params.limit * 0.4)),
  ]);

  // Merge, deduplicate by id, sort by importance DESC
  const merged = new Map<string, MemoryEntry>();
  for (const m of [...ftsResults, ...recentResults]) {
    if (!merged.has(m.id)) merged.set(m.id, m);
  }

  return [...merged.values()]
    .sort((a, b) => b.importance - a.importance)
    .slice(0, params.limit);
}

// Format memories into the injection block for the system prompt
function formatMemoryBlock(memories: MemoryEntry[]): string {
  if (memories.length === 0) return '';

  // Group by category for readability
  const byCategory = new Map<MemoryCategory, MemoryEntry[]>();
  for (const m of memories) {
    const group = byCategory.get(m.category) ?? [];
    group.push(m);
    byCategory.set(m.category, group);
  }

  const lines: string[] = ['You remember the following about this user:'];

  const categoryOrder: MemoryCategory[] = ['preference', 'goal', 'skill', 'fact', 'relationship'];
  for (const cat of categoryOrder) {
    const entries = byCategory.get(cat);
    if (!entries || entries.length === 0) continue;
    lines.push(`\n[${cat.toUpperCase()}S]`);
    for (const entry of entries) {
      lines.push(`- ${entry.content}`);
    }
  }

  return lines.join('\n');
}

export interface MemoryContextResult {
  injectionBlock: string;       // Formatted text to inject into system prompt
  memoriesUsed: MemoryEntry[];  // Which memories were injected (for access tracking)
  tokenCount: number;           // Estimated token cost of the injection block
}

// Main entry point: build memory context for injection before an inference call
export async function buildMemoryContext(params: {
  currentMessage: string;       // User's current message (used for FTS relevance)
  maxMemoryTokens: number;      // Max tokens to spend on memory injection (e.g. 400 for a 4096 ctx window)
  candidateLimit?: number;      // How many memories to fetch before token-truncating (default: 20)
}): Promise<MemoryContextResult> {
  const limit = params.candidateLimit ?? 20;
  const candidates = await selectRelevantMemories({
    query: params.currentMessage,
    limit,
  });

  // Token-budget truncation: keep highest-importance memories that fit
  const selected: MemoryEntry[] = [];
  let tokenCount = estimateTokens('You remember the following about this user:\n');

  for (const memory of candidates) {
    const lineTokens = estimateTokens(`- ${memory.content}\n`);
    if (tokenCount + lineTokens > params.maxMemoryTokens) break;
    selected.push(memory);
    tokenCount += lineTokens;
  }

  const injectionBlock = formatMemoryBlock(selected);

  return {
    injectionBlock,
    memoriesUsed: selected,
    tokenCount: estimateTokens(injectionBlock),
  };
}

// Call after inference completes to update access tracking
export async function updateAccessedMemories(memories: MemoryEntry[]): Promise<void> {
  await Promise.all(memories.map(m => memoryService.updateAccess(m.id)));
}

// Convenience: calculate the memory token budget given a companion's context window
// Reserve 15% of context window for memory injection (never more than 600 tokens)
export function calculateMemoryTokenBudget(contextWindow: number): number {
  return Math.min(Math.floor(contextWindow * 0.15), 600);
}

export class MemoryContextBuilder {
  private contextWindow: number;

  constructor(contextWindow: number) {
    this.contextWindow = contextWindow;
  }

  async build(currentMessage: string): Promise<MemoryContextResult> {
    return buildMemoryContext({
      currentMessage,
      maxMemoryTokens: calculateMemoryTokenBudget(this.contextWindow),
    });
  }

  async trackUsage(result: MemoryContextResult): Promise<void> {
    await updateAccessedMemories(result.memoriesUsed);
  }
}
```
