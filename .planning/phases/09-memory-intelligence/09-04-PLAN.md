```typescript
import { memoryService } from './memory-db';
import type { MemoryEntry } from '../types/memory';

const SUPERMEMORY_API = 'https://api.supermemory.ai/v3/memories';
const SYNC_INTERVAL_MS = 60_000;  // 60 seconds -- from ARCHITECTURE.md spec
const BATCH_LIMIT = 50;           // Max memories per sync cycle

export interface SyncCycleResult {
  synced: number;
  failed: number;
  skipped: number;  // e.g. offline
}

async function pushMemoryToSupermemory(
  memory: MemoryEntry,
  apiKey: string
): Promise<boolean> {
  try {
    const response = await fetch(SUPERMEMORY_API, {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'Authorization': `Bearer ${apiKey}`,
      },
      body: JSON.stringify({
        content: memory.content,
        metadata: {
          category: memory.category,
          importance: memory.importance,
          source: 'kin-companion',
          localId: memory.id,
          createdAt: memory.createdAt,
        },
      }),
      signal: AbortSignal.timeout(10_000),  // 10 second timeout per memory
    });

    return response.ok;
  } catch {
    return false;
  }
}

export async function runSyncCycle(apiKey: string): Promise<SyncCycleResult> {
  // Skip if offline -- navigator.onLine is available in Tauri webview
  if (typeof navigator !== 'undefined' && !navigator.onLine) {
    return { synced: 0, failed: 0, skipped: 1 };
  }

  // Skip if no API key configured
  if (!apiKey || apiKey.trim() === '') {
    return { synced: 0, failed: 0, skipped: 1 };
  }

  let unsynced: MemoryEntry[];
  try {
    unsynced = await memoryService.getUnsynced(BATCH_LIMIT);
  } catch {
    return { synced: 0, failed: 0, skipped: 0 };
  }

  if (unsynced.length === 0) {
    return { synced: 0, failed: 0, skipped: 0 };
  }

  const syncedIds: string[] = [];
  let failedCount = 0;

  // Sequential push (not parallel) to avoid rate limiting
  for (const memory of unsynced) {
    const success = await pushMemoryToSupermemory(memory, apiKey);
    if (success) {
      syncedIds.push(memory.id);
    } else {
      failedCount++;
    }
  }

  // Mark successful syncs in local DB
  if (syncedIds.length > 0) {
    try {
      await memoryService.markSynced(syncedIds);
    } catch {
      // If marking fails, memories will be re-synced next cycle -- idempotent on Supermemory side
    }
  }

  return { synced: syncedIds.length, failed: failedCount, skipped: 0 };
}

export class SupermemorySync {
  private intervalId: ReturnType<typeof setInterval> | null = null;
  private apiKey: string;

  constructor(apiKey: string) {
    this.apiKey = apiKey;
  }

  start(): void {
    // Guard against duplicate intervals
    if (this.intervalId !== null) return;

    this.intervalId = setInterval(() => {
      // Non-blocking -- interval callback does not await
      runSyncCycle(this.apiKey).catch(() => {
        // Sync cycle errors are fully silent -- user never sees sync failures
      });
    }, SYNC_INTERVAL_MS);
  }

  stop(): void {
    if (this.intervalId !== null) {
      clearInterval(this.intervalId);
      this.intervalId = null;
    }
  }

  isRunning(): boolean {
    return this.intervalId !== null;
  }

  // Trigger an immediate sync cycle outside of the interval schedule
  async flush(): Promise<SyncCycleResult> {
    return runSyncCycle(this.apiKey);
  }

  updateApiKey(newKey: string): void {
    this.apiKey = newKey;
  }
}

// Module-level singleton for convenience
let _syncInstance: SupermemorySync | null = null;

export function startMemorySync(apiKey: string): SupermemorySync {
  if (_syncInstance) {
    _syncInstance.stop();
  }
  _syncInstance = new SupermemorySync(apiKey);
  _syncInstance.start();
  return _syncInstance;
}

export function stopMemorySync(): void {
  _syncInstance?.stop();
  _syncInstance = null;
}

export function getMemorySyncInstance(): SupermemorySync | null {
  return _syncInstance;
}
```

Key design decisions:
- Sequential push (not parallel) avoids rate-limiting on the Supermemory API
- navigator.onLine check makes the service offline-safe
- Empty API key is a graceful no-op (user hasn't set up cloud sync yet)
- Sync failures are fully silent -- user experience is unaffected when offline or API is down
- markSynced failure is also handled: memories will be re-synced idempotently (Supermemory deduplicates by content hash)
- Module-level singleton pattern matches how Tauri apps typically manage background services
- flush() method allows the settings page to trigger an immediate sync on demand
