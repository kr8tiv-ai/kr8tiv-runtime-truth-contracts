The orchestrator implements the happy path from ARCHITECTURE.md section "Happy Path: User Sends Message (Local Route)" with memory steps added:

```typescript
import { streamChat } from './ollama';
import { buildSystemPrompt, getCompanionConfig } from './persona';
import { saveMessage, createConversation } from './database';
import { buildMemoryContext, updateAccessedMemories, calculateMemoryTokenBudget } from './memory-context';
import { extractMemories } from './memory-extractor';
import { memoryService } from './memory-db';
import type { Message, Conversation, StreamChunk } from '../types/chat';
import type { Bloodline } from '../types/companion';
import type { MemoryEntry } from '../types/memory';

export interface SendMessageParams {
  content: string;
  conversationId: string;
  bloodline: Bloodline;
  vramGb: number;             // For context window sizing
  onToken: (chunk: StreamChunk) => void;  // Streaming callback for UI
  onMemoriesLoaded?: (memories: MemoryEntry[]) => void;  // Optional: notify UI which memories are active
  onExtractionComplete?: (count: number) => void;        // Optional: notify UI new memories stored
}

export interface SendMessageResult {
  userMessage: Message;
  assistantMessage: Message;
  memoriesInjected: number;
  memoriesExtracted: number;
}

export async function sendMessage(params: SendMessageParams): Promise<SendMessageResult> {
  const config = getCompanionConfig(params.bloodline, params.vramGb);
  const memoryBudget = calculateMemoryTokenBudget(config.contextWindow);

  // Step 1: Fetch relevant memories BEFORE building the prompt
  const memoryContext = await buildMemoryContext({
    currentMessage: params.content,
    maxMemoryTokens: memoryBudget,
  });

  // Notify UI which memories are active (for MemoryPanel highlight)
  params.onMemoriesLoaded?.(memoryContext.memoriesUsed);

  // Step 2: Build system prompt with memory injection
  // Pass the injection block as the memories array to buildSystemPrompt
  const systemPromptWithMemory = memoryContext.injectionBlock
    ? buildSystemPrompt(config.companion, [memoryContext.injectionBlock])
    : buildSystemPrompt(config.companion);

  const configWithMemory = { ...config, systemPrompt: systemPromptWithMemory };

  // Step 3: Save user message to DB
  const userMessage = await saveMessage(
    params.conversationId,
    'user',
    params.content,
    'user-input',
    0
  );

  // Step 4: Stream inference (the LLM does its thing)
  let fullResponse = '';
  let tokensUsed = 0;

  // Load conversation history for context (last 20 messages to stay within context window)
  const { getMessages } = await import('./database');
  const history = await getMessages(params.conversationId);
  const recentHistory = history.slice(-20).map(m => ({
    role: m.role as 'user' | 'assistant' | 'system',
    content: m.content,
  }));

  const stream = streamChat(configWithMemory, recentHistory);
  for await (const chunk of stream) {
    if (!chunk.done) {
      fullResponse += chunk.content;
      params.onToken(chunk);
    } else {
      tokensUsed = chunk.tokensGenerated ?? 0;
    }
  }

  // Step 5: Save assistant message to DB
  const assistantMessage = await saveMessage(
    params.conversationId,
    'assistant',
    fullResponse,
    `local:${config.companion.modelId}`,
    tokensUsed
  );

  // Step 6: Update memory access tracking (which memories were used)
  await updateAccessedMemories(memoryContext.memoriesUsed);

  // Step 7: Extract new memories from this exchange (async, non-blocking to response)
  // Fire and forget -- extraction result is reported via callback but doesn't block return
  extractMemories({
    userMessage: params.content,
    assistantMessage: fullResponse,
    sourceMessageId: assistantMessage.id,
    conversationId: params.conversationId,
    modelId: config.companion.modelId,
  }).then(result => {
    params.onExtractionComplete?.(result.extracted.length);
  }).catch(() => {
    // Extraction failure is silent -- never crash the chat
  });

  return {
    userMessage,
    assistantMessage,
    memoriesInjected: memoryContext.memoriesUsed.length,
    memoriesExtracted: 0, // Set async via onExtractionComplete callback
  };
}

export class ChatOrchestrator {
  private bloodline: Bloodline;
  private vramGb: number;

  constructor(bloodline: Bloodline, vramGb: number) {
    this.bloodline = bloodline;
    this.vramGb = vramGb;
  }

  async send(params: Omit<SendMessageParams, 'bloodline' | 'vramGb'>): Promise<SendMessageResult> {
    return sendMessage({ ...params, bloodline: this.bloodline, vramGb: this.vramGb });
  }
}
```

A collapsible side panel that shows what the companion remembers. Grouped by category, sortable by importance, with delete buttons.

```svelte
<script lang="ts">
  import { onMount } from 'svelte';
  import { memoryService } from '$lib/services/memory-db';
  import type { MemoryEntry, MemoryCategory } from '$lib/types/memory';

  let memories: MemoryEntry[] = [];
  let isLoading = true;
  let activeMemoryIds: Set<string> = new Set(); // highlighted during active inference
  let filterCategory: MemoryCategory | 'all' = 'all';

  export function highlightMemories(ids: string[]) {
    activeMemoryIds = new Set(ids);
    // Clear highlight after 3 seconds
    setTimeout(() => { activeMemoryIds = new Set(); }, 3000);
  }

  async function loadMemories() {
    isLoading = true;
    memories = await memoryService.getRecent(50);
    isLoading = false;
  }

  async function deleteMemory(id: string) {
    await memoryService.delete(id);
    memories = memories.filter(m => m.id !== id);
  }

  export async function refresh() {
    await loadMemories();
  }

  onMount(loadMemories);

  $: filtered = filterCategory === 'all'
    ? memories
    : memories.filter(m => m.category === filterCategory);

  const categoryColors: Record<MemoryCategory, string> = {
    fact: '#5BC0EB',
    preference: '#6C63FF',
    skill: '#00D4AA',
    relationship: '#FFB347',
    goal: '#E88AFF',
  };

  function formatDate(iso: string): string {
    return new Date(iso).toLocaleDateString(undefined, { month: 'short', day: 'numeric' });
  }
</script>

<div class="memory-panel">
  <div class="panel-header">
    <h3>What I Remember</h3>
    <span class="memory-count">{memories.length} memories</span>
  </div>

  <div class="category-filters">
    {#each ['all', 'preference', 'goal', 'skill', 'fact', 'relationship'] as cat}
      <button
        class="filter-chip"
        class:active={filterCategory === cat}
        on:click={() => filterCategory = cat as MemoryCategory | 'all'}
      >
        {cat}
      </button>
    {/each}
  </div>

  {#if isLoading}
    <div class="loading">Loading memories...</div>
  {:else if filtered.length === 0}
    <div class="empty-state">
      <p>No memories yet.</p>
      <p class="hint">I'll start remembering things about you as we talk.</p>
    </div>
  {:else}
    <ul class="memory-list">
      {#each filtered as memory (memory.id)}
        <li
          class="memory-item"
          class:active={activeMemoryIds.has(memory.id)}
          style="--category-color: {categoryColors[memory.category]}"
        >
          <div class="memory-content">
            <span class="category-badge" style="color: {categoryColors[memory.category]}">
              {memory.category}
            </span>
            <p class="memory-text">{memory.content}</p>
            <div class="memory-meta">
              <span class="importance" title="Importance">
                {'●'.repeat(Math.ceil(memory.importance * 5))}{'○'.repeat(5 - Math.ceil(memory.importance * 5))}
              </span>
              <span class="date">{formatDate(memory.lastAccessed)}</span>
            </div>
          </div>
          <button
            class="delete-btn"
            on:click={() => deleteMemory(memory.id)}
            title="Forget this"
            aria-label="Delete memory"
          >×</button>
        </li>
      {/each}
    </ul>
  {/if}
</div>
```
