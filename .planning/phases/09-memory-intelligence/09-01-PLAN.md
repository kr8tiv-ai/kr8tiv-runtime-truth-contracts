```typescript
export type MemoryCategory = 'fact' | 'preference' | 'skill' | 'relationship' | 'goal';

export interface MemoryEntry {
  id: string;                    // UUID v4
  content: string;               // The remembered information (plain English sentence)
  category: MemoryCategory;
  importance: number;            // 0.0 to 1.0 -- higher = more critical to inject into context
  embedding: number[] | null;    // Local embedding vector from Ollama /api/embeddings (null until generated)
  sourceMessageId: string | null; // Which message triggered this memory extraction
  conversationId: string | null;  // Which conversation this came from
  createdAt: string;             // ISO 8601
  lastAccessed: string;          // ISO 8601 -- updated every time memory is injected into context
  accessCount: number;           // How many times this memory has been used
  syncedAt: string | null;       // ISO 8601 -- null = not yet synced to Supermemory
  localOnly: boolean;            // User-marked as private -- never sync to cloud
}

export interface MemorySearchResult {
  memory: MemoryEntry;
  score: number;                 // Relevance score (0.0 to 1.0) from FTS or embedding similarity
  matchType: 'fts' | 'embedding' | 'category';
}

export interface MemoryStore {
  entries: MemoryEntry[];
  totalCount: number;
  lastSyncedAt: string | null;
}

export interface MemoryExtractionResult {
  extracted: MemoryEntry[];      // New memories found in the message
  updated: MemoryEntry[];        // Existing memories that were reinforced/updated
  skipped: number;               // How many candidates were rejected (duplicates/low importance)
}
```

Schema (add to the existing database init):
```sql
CREATE TABLE IF NOT EXISTS memories (
  id TEXT PRIMARY KEY,
  content TEXT NOT NULL,
  category TEXT NOT NULL CHECK(category IN ('fact','preference','skill','relationship','goal')),
  importance REAL NOT NULL DEFAULT 0.5,
  embedding BLOB,                           -- NULL until Ollama embedding generated
  source_message_id TEXT,
  conversation_id TEXT,
  created_at TEXT NOT NULL,
  last_accessed TEXT NOT NULL,
  access_count INTEGER NOT NULL DEFAULT 0,
  synced_at TEXT,                           -- NULL = pending sync
  local_only INTEGER NOT NULL DEFAULT 0     -- SQLite boolean
);

CREATE INDEX IF NOT EXISTS idx_memories_category ON memories(category, importance DESC);
CREATE INDEX IF NOT EXISTS idx_memories_last_accessed ON memories(last_accessed DESC);
CREATE INDEX IF NOT EXISTS idx_memories_sync ON memories(synced_at) WHERE synced_at IS NULL;

-- FTS5 virtual table for full-text search over memory content
CREATE VIRTUAL TABLE IF NOT EXISTS memories_fts USING fts5(
  content,
  content=memories,
  content_rowid=rowid
);

-- Triggers to keep FTS5 in sync
CREATE TRIGGER IF NOT EXISTS memories_fts_insert AFTER INSERT ON memories BEGIN
  INSERT INTO memories_fts(rowid, content) VALUES (new.rowid, new.content);
END;

CREATE TRIGGER IF NOT EXISTS memories_fts_delete AFTER DELETE ON memories BEGIN
  INSERT INTO memories_fts(memories_fts, rowid, content) VALUES('delete', old.rowid, old.content);
END;

CREATE TRIGGER IF NOT EXISTS memories_fts_update AFTER UPDATE ON memories BEGIN
  INSERT INTO memories_fts(memories_fts, rowid, content) VALUES('delete', old.rowid, old.content);
  INSERT INTO memories_fts(rowid, content) VALUES (new.rowid, new.content);
END;
```

Tauri commands (all in memory.rs):

1. `#[tauri::command] pub fn mem_store(state: tauri::State<DbState>, content: String, category: String, importance: f64, source_message_id: Option<String>, conversation_id: Option<String>, local_only: bool) -> Result<String, String>`
   - Generate UUID v4 for id (use `uuid` crate: `uuid = { version = "1", features = ["v4"] }`)
   - Set created_at and last_accessed to current UTC timestamp as ISO 8601 string
   - INSERT into memories table
   - Return JSON-serialized MemoryEntry via serde_json

2. `#[tauri::command] pub fn mem_search_by_category(state: tauri::State<DbState>, category: String, limit: i64) -> Result<String, String>`
   - SELECT from memories WHERE category = ?1 ORDER BY importance DESC LIMIT ?2
   - Return JSON array of MemoryEntry

3. `#[tauri::command] pub fn mem_get_recent(state: tauri::State<DbState>, limit: i64) -> Result<String, String>`
   - SELECT from memories ORDER BY last_accessed DESC LIMIT ?1
   - Return JSON array of MemoryEntry

4. `#[tauri::command] pub fn mem_update_access(state: tauri::State<DbState>, id: String) -> Result<(), String>`
   - UPDATE memories SET access_count = access_count + 1, last_accessed = ?1 WHERE id = ?2
   - Return Ok(()) on success, Err with message if not found

5. `#[tauri::command] pub fn mem_delete(state: tauri::State<DbState>, id: String) -> Result<(), String>`
   - DELETE FROM memories WHERE id = ?1
   - Return Ok(()) -- no error if id not found (idempotent)

6. `#[tauri::command] pub fn mem_search_fts(state: tauri::State<DbState>, query: String, limit: i64) -> Result<String, String>`
   - SELECT m.* FROM memories m JOIN memories_fts f ON m.rowid = f.rowid WHERE memories_fts MATCH ?1 ORDER BY rank LIMIT ?2
   - Return JSON array of MemoryEntry

7. `#[tauri::command] pub fn mem_get_unsynced(state: tauri::State<DbState>, limit: i64) -> Result<String, String>`
   - SELECT * FROM memories WHERE synced_at IS NULL AND local_only = 0 ORDER BY created_at ASC LIMIT ?1
   - Return JSON array of MemoryEntry (for the sync service to consume)

8. `#[tauri::command] pub fn mem_mark_synced(state: tauri::State<DbState>, ids: Vec<String>) -> Result<(), String>`
   - UPDATE memories SET synced_at = ?1 WHERE id IN (ids)
   - Use a parameterized query with repeated placeholders for the IN clause
   - Return Ok(()) on success

```typescript
import { invoke } from '@tauri-apps/api/core';
import type { MemoryEntry, MemoryCategory, MemorySearchResult } from '../types/memory';

export class MemoryService {
  async store(params: {
    content: string;
    category: MemoryCategory;
    importance: number;
    sourceMessageId?: string;
    conversationId?: string;
    localOnly?: boolean;
  }): Promise<MemoryEntry> {
    const json = await invoke<string>('mem_store', {
      content: params.content,
      category: params.category,
      importance: params.importance,
      sourceMessageId: params.sourceMessageId ?? null,
      conversationId: params.conversationId ?? null,
      localOnly: params.localOnly ?? false,
    });
    return JSON.parse(json) as MemoryEntry;
  }

  async searchByCategory(category: MemoryCategory, limit = 20): Promise<MemoryEntry[]> {
    const json = await invoke<string>('mem_search_by_category', { category, limit });
    return JSON.parse(json) as MemoryEntry[];
  }

  async getRecent(limit = 10): Promise<MemoryEntry[]> {
    const json = await invoke<string>('mem_get_recent', { limit });
    return JSON.parse(json) as MemoryEntry[];
  }

  async updateAccess(id: string): Promise<void> {
    await invoke<void>('mem_update_access', { id });
  }

  async delete(id: string): Promise<void> {
    await invoke<void>('mem_delete', { id });
  }

  async searchFts(query: string, limit = 10): Promise<MemoryEntry[]> {
    const json = await invoke<string>('mem_search_fts', { query, limit });
    return JSON.parse(json) as MemoryEntry[];
  }

  async getUnsynced(limit = 100): Promise<MemoryEntry[]> {
    const json = await invoke<string>('mem_get_unsynced', { limit });
    return JSON.parse(json) as MemoryEntry[];
  }

  async markSynced(ids: string[]): Promise<void> {
    await invoke<void>('mem_mark_synced', { ids });
  }
}

export const memoryService = new MemoryService();
```
