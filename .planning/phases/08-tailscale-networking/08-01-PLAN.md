export type ConnectionState =
  | 'checking'        // Initial state -- detecting Tailscale
  | 'not_installed'   // Tailscale CLI not found on PATH
  | 'not_authenticated' // CLI present but not logged in
  | 'authenticating'  // Auth flow started, waiting for user to visit auth URL
  | 'connected'       // Authenticated and connected to tailnet
  | 'degraded'        // Connected but relay only (no direct peer -- DERP relay)
  | 'local_only';     // Tailscale unavailable -- operating in local-only mode

export interface TailscaleDevice {
  hostname: string;         // e.g. "kin-companion"
  tailscaleIp: string;      // 100.x.x.x address
  magicdnsName: string;     // e.g. "kin-companion.tailnet.ts.net"
  isOnline: boolean;
  os: string;
}

export interface NetworkStatus {
  state: ConnectionState;
  device: TailscaleDevice | null;   // null when not connected
  tailscaleVersion: string | null;  // CLI version, null if not installed
  authUrl: string | null;           // Set during 'authenticating' state
  error: string | null;             // User-friendly error message
  // Service reachability (checked by pinging the tailnet services)
  frontierReachable: boolean;       // frontier.kr8tiv.ts.net
  syncReachable: boolean;           // sync.kr8tiv.ts.net
}

export const DEFAULT_NETWORK_STATUS: NetworkStatus = {
  state: 'checking',
  device: null,
  tailscaleVersion: null,
  authUrl: null,
  error: null,
  frontierReachable: false,
  syncReachable: false,
};

use serde::{Deserialize, Serialize};
use tokio::process::Command;

// --- Types mirroring TypeScript types/network.ts ---

#[derive(Clone, Serialize, Deserialize, Debug)]
pub struct TailscaleDevice {
    pub hostname: String,
    pub tailscale_ip: String,
    pub magicdns_name: String,
    pub is_online: bool,
    pub os: String,
}

#[derive(Clone, Serialize, Debug)]
pub struct TailscaleStatusResult {
    pub installed: bool,
    pub version: Option<String>,
    pub authenticated: bool,
    pub connected: bool,
    pub device: Option<TailscaleDevice>,
    pub backend_state: Option<String>,  // Tailscale's own state string
}

// --- Tailscale CLI version of status JSON (partial) ---
#[derive(Deserialize, Debug)]
struct TsStatusJson {
    #[serde(rename = "BackendState")]
    backend_state: Option<String>,
    #[serde(rename = "Self")]
    self_node: Option<TsNodeJson>,
    #[serde(rename = "Version")]
    version: Option<String>,
}

#[derive(Deserialize, Debug)]
struct TsNodeJson {
    #[serde(rename = "HostName")]
    hostname: Option<String>,
    #[serde(rename = "TailscaleIPs")]
    tailscale_ips: Option<Vec<String>>,
    #[serde(rename = "DNSName")]
    dns_name: Option<String>,
    #[serde(rename = "Online")]
    online: Option<bool>,
    #[serde(rename = "OS")]
    os: Option<String>,
}

// --- Tauri Commands ---

/// Check if Tailscale is installed and get its status.
/// Returns full status JSON. Never errors -- missing Tailscale returns installed=false.
#[tauri::command]
pub async fn tailscale_status() -> Result<String, String> {
    // Check if tailscale binary is on PATH
    let version_check = Command::new("tailscale")
        .args(["version"])
        .output()
        .await;

    let installed = version_check.is_ok()
        && version_check.as_ref().map(|o| o.status.success()).unwrap_or(false);

    if !installed {
        let result = TailscaleStatusResult {
            installed: false,
            version: None,
            authenticated: false,
            connected: false,
            device: None,
            backend_state: None,
        };
        return Ok(serde_json::to_string(&result).unwrap());
    }

    let version = version_check
        .ok()
        .and_then(|o| String::from_utf8(o.stdout).ok())
        .map(|s| s.trim().to_string());

    // Run `tailscale status --json` for machine-readable output
    let status_output = Command::new("tailscale")
        .args(["status", "--json"])
        .output()
        .await
        .map_err(|e| format!("tailscale status failed: {}", e))?;

    let status_str = String::from_utf8_lossy(&status_output.stdout);
    let ts_status: Option<TsStatusJson> = serde_json::from_str(&status_str).ok();

    let backend_state = ts_status
        .as_ref()
        .and_then(|s| s.backend_state.clone());

    // BackendState values: "Running" = connected, "NeedsLogin" = not authed,
    // "NoState" = not initialized, "Stopped" = stopped
    let authenticated = backend_state
        .as_deref()
        .map(|s| s != "NeedsLogin" && s != "NoState" && s != "Stopped")
        .unwrap_or(false);

    let connected = backend_state.as_deref() == Some("Running");

    let device = ts_status.as_ref().and_then(|s| s.self_node.as_ref()).map(|node| {
        let ip = node
            .tailscale_ips
            .as_ref()
            .and_then(|ips| ips.first())
            .cloned()
            .unwrap_or_default();
        // Tailscale DNSName ends with a dot (e.g. "host.tailnet.ts.net.")
        let dns = node
            .dns_name
            .clone()
            .unwrap_or_default()
            .trim_end_matches('.')
            .to_string();
        TailscaleDevice {
            hostname: node.hostname.clone().unwrap_or_default(),
            tailscale_ip: ip,
            magicdns_name: dns,
            is_online: node.online.unwrap_or(false),
            os: node.os.clone().unwrap_or_default(),
        }
    });

    let result = TailscaleStatusResult {
        installed: true,
        version,
        authenticated,
        connected,
        device,
        backend_state,
    };

    Ok(serde_json::to_string(&result).unwrap())
}

/// Start Tailscale authentication flow.
/// Returns the auth URL the user must visit in their browser.
/// After visiting, call tailscale_status() to confirm connected.
#[tauri::command]
pub async fn tailscale_start_auth(
    app: tauri::AppHandle,
    hostname: String,   // e.g. "kin-companion"
) -> Result<String, String> {
    // `tailscale up --hostname {hostname} --accept-routes` starts the auth flow.
    // When not yet authenticated, it prints an auth URL to stdout.
    // We capture that URL and return it.
    let output = Command::new("tailscale")
        .args([
            "up",
            "--hostname", &hostname,
            "--accept-routes",
            "--accept-dns=true",
        ])
        .output()
        .await
        .map_err(|e| format!("Failed to run tailscale up: {}. Is Tailscale installed?", e))?;

    let stdout = String::from_utf8_lossy(&output.stdout).to_string();
    let stderr = String::from_utf8_lossy(&output.stderr).to_string();
    let combined = format!("{}{}", stdout, stderr);

    // Auth URL pattern: "https://login.tailscale.com/a/..."
    if let Some(url_start) = combined.find("https://login.tailscale.com") {
        let url = combined[url_start..]
            .split_whitespace()
            .next()
            .unwrap_or("")
            .to_string();
        log::info!("Tailscale auth URL: {}", url);
        return Ok(url);
    }

    // If output contains "Success" or already running, return empty string (already authed)
    if combined.contains("Success") || output.status.success() {
        log::info!("Tailscale already authenticated or successfully connected");
        return Ok(String::new());
    }

    Err(format!(
        "Tailscale auth failed. Check Tailscale is installed and try again. Output: {}",
        combined.trim()
    ))
}

/// Disconnect from Tailscale (graceful -- preserves auth credentials).
#[tauri::command]
pub async fn tailscale_down() -> Result<String, String> {
    let output = Command::new("tailscale")
        .args(["down"])
        .output()
        .await
        .map_err(|e| format!("tailscale down failed: {}", e))?;

    if output.status.success() {
        log::info!("Tailscale disconnected");
        Ok("disconnected".to_string())
    } else {
        let stderr = String::from_utf8_lossy(&output.stderr).to_string();
        Err(format!("tailscale down error: {}", stderr.trim()))
    }
}

/// Ping a Tailscale service hostname to check reachability.
/// Returns true if reachable within 3 seconds.
#[tauri::command]
pub async fn tailscale_ping(hostname: String) -> Result<bool, String> {
    // Use tailscale ping which does a direct WireGuard ping (not ICMP)
    let output = tokio::time::timeout(
        std::time::Duration::from_secs(5),
        Command::new("tailscale")
            .args(["ping", "--timeout=3s", "--c=1", &hostname])
            .output(),
    )
    .await
    .map_err(|_| "ping timeout".to_string())?
    .map_err(|e| format!("ping failed: {}", e))?;

    Ok(output.status.success())
}

import { invoke } from '@tauri-apps/api/core';
import type { NetworkStatus, TailscaleDevice, ConnectionState } from '../types/network';

// Shape returned by tailscale_status Rust command (snake_case from Rust)
interface RustStatusResult {
  installed: boolean;
  version: string | null;
  authenticated: boolean;
  connected: boolean;
  device: {
    hostname: string;
    tailscale_ip: string;
    magicdns_name: string;
    is_online: boolean;
    os: string;
  } | null;
  backend_state: string | null;
}

export async function getTailscaleStatus(): Promise<NetworkStatus> {
  try {
    const raw = await invoke<string>('tailscale_status');
    const result: RustStatusResult = JSON.parse(raw);

    let state: ConnectionState;
    if (!result.installed) {
      state = 'not_installed';
    } else if (!result.authenticated) {
      state = 'not_authenticated';
    } else if (!result.connected) {
      state = 'not_authenticated'; // authenticated but not up
    } else {
      state = 'connected';
    }

    const device: TailscaleDevice | null = result.device
      ? {
          hostname: result.device.hostname,
          tailscaleIp: result.device.tailscale_ip,
          magicdnsName: result.device.magicdns_name,
          isOnline: result.device.is_online,
          os: result.device.os,
        }
      : null;

    return {
      state,
      device,
      tailscaleVersion: result.version,
      authUrl: null,
      error: null,
      frontierReachable: false,
      syncReachable: false,
    };
  } catch (err) {
    // Failed to run command -- Tauri IPC error
    return {
      state: 'local_only',
      device: null,
      tailscaleVersion: null,
      authUrl: null,
      error: `Failed to check Tailscale status: ${err}`,
      frontierReachable: false,
      syncReachable: false,
    };
  }
}

/**
 * Start the Tailscale auth flow for the KIN companion.
 * Returns the auth URL the user must visit to authenticate.
 * Returns null if already authenticated.
 */
export async function startTailscaleAuth(hostname = 'kin-companion'): Promise<string | null> {
  const url = await invoke<string>('tailscale_start_auth', { hostname });
  return url || null; // empty string = already authed
}

/**
 * Disconnect from Tailscale (keeps credentials, just stops the tunnel).
 */
export async function disconnectTailscale(): Promise<void> {
  await invoke<string>('tailscale_down');
}

/**
 * Check if a Tailscale service hostname is reachable.
 * Used to verify kr8tiv service connectivity.
 */
export async function pingTailscaleHost(hostname: string): Promise<boolean> {
  try {
    return await invoke<boolean>('tailscale_ping', { hostname });
  } catch {
    return false;
  }
}

/**
 * Check reachability of both kr8tiv services.
 * Non-throwing -- returns false on any error.
 */
export async function checkKr8tivServices(): Promise<{
  frontierReachable: boolean;
  syncReachable: boolean;
}> {
  const [frontierReachable, syncReachable] = await Promise.allSettled([
    pingTailscaleHost('frontier.kr8tiv.ts.net'),
    pingTailscaleHost('sync.kr8tiv.ts.net'),
  ]).then((results) =>
    results.map((r) => (r.status === 'fulfilled' ? r.value : false))
  );

  return {
    frontierReachable: frontierReachable as boolean,
    syncReachable: syncReachable as boolean,
  };
}

import { writable, derived, get } from 'svelte/store';
import type { NetworkStatus } from '../types/network';
import { DEFAULT_NETWORK_STATUS } from '../types/network';
import {
  getTailscaleStatus,
  startTailscaleAuth,
  disconnectTailscale,
  checkKr8tivServices,
} from '../services/tailscale';

// Core network status store
export const networkStore = writable<NetworkStatus>(DEFAULT_NETWORK_STATUS);

// Derived convenience booleans
export const isTailscaleConnected = derived(
  networkStore,
  ($n) => $n.state === 'connected' || $n.state === 'degraded'
);

export const isLocalOnly = derived(
  networkStore,
  ($n) => $n.state === 'local_only' || $n.state === 'not_installed'
);

export const canReachFrontier = derived(
  networkStore,
  ($n) => $n.frontierReachable
);

/**
 * Refresh Tailscale status and update the store.
 * Called on app startup and periodically (every 60 seconds).
 * Never throws -- always falls back to local_only on error.
 */
export async function refreshNetworkStatus(): Promise<void> {
  networkStore.update((s) => ({ ...s, state: 'checking' }));

  try {
    const status = await getTailscaleStatus();

    // If connected, also check service reachability
    if (status.state === 'connected') {
      const services = await checkKr8tivServices();
      networkStore.set({ ...status, ...services });
    } else {
      networkStore.set(status);
    }
  } catch (err) {
    networkStore.set({
      ...DEFAULT_NETWORK_STATUS,
      state: 'local_only',
      error: `Network check failed: ${err}`,
    });
  }
}

/**
 * Initiate Tailscale authentication flow.
 * Updates store state to 'authenticating' and sets the auth URL.
 * Caller is responsible for opening the URL in the user's browser.
 */
export async function initiateAuth(): Promise<string | null> {
  networkStore.update((s) => ({ ...s, state: 'authenticating', authUrl: null, error: null }));

  try {
    const authUrl = await startTailscaleAuth('kin-companion');
    if (authUrl) {
      networkStore.update((s) => ({ ...s, authUrl }));
      return authUrl;
    } else {
      // Already authenticated -- refresh to confirm
      await refreshNetworkStatus();
      return null;
    }
  } catch (err) {
    networkStore.update((s) => ({
      ...s,
      state: 'local_only',
      error: `Authentication failed: ${err}`,
    }));
    return null;
  }
}

/**
 * Disconnect from Tailscale and switch to local-only mode.
 */
export async function switchToLocalOnly(): Promise<void> {
  try {
    await disconnectTailscale();
  } catch {
    // Ignore disconnect errors -- user wants local-only regardless
  }
  networkStore.set({
    ...DEFAULT_NETWORK_STATUS,
