```rust
use serde::{Deserialize, Serialize};
use std::process::Command;

#[derive(Debug, Serialize, Deserialize, Clone)]
pub struct MemoryStats {
    pub system_ram_total_gb: f32,
    pub system_ram_used_gb: f32,
    pub system_ram_percent: f32,
    pub vram_total_gb: f32,
    pub vram_used_gb: f32,
    pub vram_percent: f32,
    pub ollama_rss_mb: f32,       // Ollama process resident set size
}

/// Get system memory stats.
/// VRAM via nvidia-smi. System RAM via platform commands.
/// Ollama RSS via process list.
#[tauri::command]
pub async fn get_memory_stats() -> Result<MemoryStats, String> {
    let mut stats = MemoryStats {
        system_ram_total_gb: 0.0,
        system_ram_used_gb: 0.0,
        system_ram_percent: 0.0,
        vram_total_gb: 0.0,
        vram_used_gb: 0.0,
        vram_percent: 0.0,
        ollama_rss_mb: 0.0,
    };

    // VRAM: nvidia-smi (Windows/Linux)
    if let Ok(output) = Command::new("nvidia-smi")
        .args(["--query-gpu=memory.total,memory.used", "--format=csv,noheader,nounits"])
        .output()
    {
        if output.status.success() {
            let text = String::from_utf8_lossy(&output.stdout);
            let parts: Vec<&str> = text.trim().splitn(2, ',').collect();
            if parts.len() == 2 {
                let total_mb: f32 = parts[0].trim().parse().unwrap_or(0.0);
                let used_mb: f32 = parts[1].trim().parse().unwrap_or(0.0);
                stats.vram_total_gb = total_mb / 1024.0;
                stats.vram_used_gb = used_mb / 1024.0;
                stats.vram_percent = if total_mb > 0.0 { (used_mb / total_mb) * 100.0 } else { 0.0 };
            }
        }
    }

    // System RAM: Windows WMIC
    #[cfg(target_os = "windows")]
    {
        if let Ok(total_out) = Command::new("wmic")
            .args(["OS", "get", "TotalVisibleMemorySize", "/value"])
            .output()
        {
            let text = String::from_utf8_lossy(&total_out.stdout);
            if let Some(line) = text.lines().find(|l| l.starts_with("TotalVisibleMemorySize=")) {
                let kb: f32 = line.trim_start_matches("TotalVisibleMemorySize=").trim().parse().unwrap_or(0.0);
                stats.system_ram_total_gb = kb / (1024.0 * 1024.0);
            }
        }
        if let Ok(free_out) = Command::new("wmic")
            .args(["OS", "get", "FreePhysicalMemory", "/value"])
            .output()
        {
            let text = String::from_utf8_lossy(&free_out.stdout);
            if let Some(line) = text.lines().find(|l| l.starts_with("FreePhysicalMemory=")) {
                let kb: f32 = line.trim_start_matches("FreePhysicalMemory=").trim().parse().unwrap_or(0.0);
                let free_gb = kb / (1024.0 * 1024.0);
                stats.system_ram_used_gb = stats.system_ram_total_gb - free_gb;
                if stats.system_ram_total_gb > 0.0 {
                    stats.system_ram_percent = (stats.system_ram_used_gb / stats.system_ram_total_gb) * 100.0;
                }
            }
        }
    }

    // System RAM: macOS vm_stat
    #[cfg(target_os = "macos")]
    {
        if let Ok(output) = Command::new("vm_stat").output() {
            let text = String::from_utf8_lossy(&output.stdout);
            // vm_stat returns page counts; page size is 4096 bytes
            let get_pages = |key: &str| -> f32 {
                text.lines()
                    .find(|l| l.contains(key))
                    .and_then(|l| l.split(':').nth(1))
                    .and_then(|v| v.trim().trim_end_matches('.').parse::<f32>().ok())
                    .unwrap_or(0.0)
            };
            let free = get_pages("Pages free");
            let active = get_pages("Pages active");
            let inactive = get_pages("Pages inactive");
            let wired = get_pages("Pages wired down");
            let page_size = 4096.0_f32;
            let total_pages = free + active + inactive + wired;
            stats.system_ram_total_gb = (total_pages * page_size) / (1024.0_f32.powi(3));
            stats.system_ram_used_gb = ((active + inactive + wired) * page_size) / (1024.0_f32.powi(3));
            if total_pages > 0.0 {
                stats.system_ram_percent = ((active + inactive + wired) / total_pages) * 100.0;
            }
        }
    }

    Ok(stats)
}

/// Get Ollama process RSS (resident set size in MB) by scanning process list.
/// Returns 0 if Ollama process not found.
#[tauri::command]
pub async fn get_ollama_memory() -> Result<f32, String> {
    // Windows: tasklist /FO CSV /FI "IMAGENAME eq ollama.exe"
    #[cfg(target_os = "windows")]
    {
        if let Ok(output) = Command::new("tasklist")
            .args(["/FO", "CSV", "/FI", "IMAGENAME eq ollama.exe"])
            .output()
        {
            let text = String::from_utf8_lossy(&output.stdout);
            // CSV format: "Image Name","PID","Session Name","Session#","Mem Usage"
            for line in text.lines().skip(1) {
                if line.contains("ollama") {
                    let parts: Vec<&str> = line.split(',').collect();
                    if parts.len() >= 5 {
                        // Mem Usage field: "XXX,XXX K" -- strip quotes, commas, K
                        let mem_str = parts[4].trim_matches('"').replace(',', "").replace(" K", "");
                        let kb: f32 = mem_str.trim().parse().unwrap_or(0.0);
                        return Ok(kb / 1024.0);
                    }
                }
            }
        }
    }

    // macOS/Linux: ps aux | grep ollama
    #[cfg(not(target_os = "windows"))]
    {
        if let Ok(output) = Command::new("ps")
            .args(["aux"])
            .output()
        {
            let text = String::from_utf8_lossy(&output.stdout);
            for line in text.lines() {
                if line.contains("ollama serve") || line.contains("ollama") {
                    let parts: Vec<&str> = line.split_whitespace().collect();
                    if parts.len() >= 6 {
                        // ps aux col 5 = VSZ (KB), col 6 = RSS (KB)
                        let rss_kb: f32 = parts[5].parse().unwrap_or(0.0);
                        return Ok(rss_kb / 1024.0);
                    }
                }
            }
        }
    }

    Ok(0.0)
}
```

```typescript
import { invoke } from '@tauri-apps/api/core';

// Thresholds from PITFALLS.md Pitfall 3
const VRAM_RESTART_THRESHOLD = 90;       // % -- restart before OOM
const RAM_RESTART_THRESHOLD = 85;        // % -- restart before system slowdown
const POLL_INTERVAL_MS = 30_000;         // 30 seconds
const MAX_INTERACTIONS_BEFORE_RESTART = 200; // Force restart after N messages regardless

export interface MemoryStats {
  system_ram_total_gb: number;
  system_ram_used_gb: number;
  system_ram_percent: number;
  vram_total_gb: number;
  vram_used_gb: number;
  vram_percent: number;
  ollama_rss_mb: number;
}

export interface RestartEvent {
  timestamp: string;           // ISO 8601
  reason: 'vram_threshold' | 'ram_threshold' | 'interaction_limit' | 'manual';
  vram_percent: number;
  ram_percent: number;
  ollama_rss_mb: number;
  interaction_count: number;
}

let pollTimer: ReturnType<typeof setInterval> | null = null;
let interactionCount = 0;
let restartLog: RestartEvent[] = [];
let isRestarting = false;
let onRestartCallback: ((event: RestartEvent) => void) | null = null;

export function onRestart(cb: (event: RestartEvent) => void): void {
  onRestartCallback = cb;
}

export function incrementInteractions(): void {
  interactionCount++;
}

export function getRestartLog(): RestartEvent[] {
  return [...restartLog];
}

async function checkAndMaybeRestart(): Promise<void> {
  if (isRestarting) return;

  let stats: MemoryStats;
  try {
    stats = await invoke<MemoryStats>('get_memory_stats');
  } catch {
    return; // Skip if command fails
  }

  const vramHigh = stats.vram_total_gb > 0 && stats.vram_percent >= VRAM_RESTART_THRESHOLD;
  const ramHigh = stats.system_ram_total_gb > 0 && stats.system_ram_percent >= RAM_RESTART_THRESHOLD;
  const tooManyInteractions = interactionCount >= MAX_INTERACTIONS_BEFORE_RESTART;

  if (!vramHigh && !ramHigh && !tooManyInteractions) return;

  const reason: RestartEvent['reason'] = vramHigh
    ? 'vram_threshold'
    : ramHigh
    ? 'ram_threshold'
    : 'interaction_limit';

  await triggerGracefulRestart(reason, stats);
}

async function triggerGracefulRestart(
  reason: RestartEvent['reason'],
  stats: MemoryStats
): Promise<void> {
  isRestarting = true;
  const event: RestartEvent = {
    timestamp: new Date().toISOString(),
    reason,
    vram_percent: stats.vram_percent,
    ram_percent: stats.system_ram_percent,
    ollama_rss_mb: stats.ollama_rss_mb,
    interaction_count: interactionCount,
  };

  try {
    // 1. Stop Ollama
    await invoke('stop_ollama');
    // 2. Brief pause to let memory free
    await new Promise(r => setTimeout(r, 2000));
    // 3. Restart Ollama (start_ollama polls health and waits up to 10s)
    await invoke('start_ollama');
    // 4. Reset interaction counter
    interactionCount = 0;
    // 5. Log event
    restartLog = [event, ...restartLog].slice(0, 50); // Keep last 50 restart events
    // 6. Notify UI
    onRestartCallback?.(event);
  } catch (e) {
    console.error('Graceful restart failed:', e);
  } finally {
    isRestarting = false;
  }
}

export function startMemoryWatch(): void {
  if (pollTimer) return; // Already watching
  pollTimer = setInterval(checkAndMaybeRestart, POLL_INTERVAL_MS);
  // Also run immediately on start
  checkAndMaybeRestart();
}

export function stopMemoryWatch(): void {
  if (pollTimer) {
    clearInterval(pollTimer);
    pollTimer = null;
  }
}

export async function triggerManualRestart(): Promise<void> {
  let stats: MemoryStats;
  try {
    stats = await invoke<MemoryStats>('get_memory_stats');
  } catch {
    stats = { system_ram_total_gb: 0, system_ram_used_gb: 0, system_ram_percent: 0, vram_total_gb: 0, vram_used_gb: 0, vram_percent: 0, ollama_rss_mb: 0 };
  }
  await triggerGracefulRestart('manual', stats);
}

export async function getCurrentMemoryStats(): Promise<MemoryStats | null> {
  try {
    return await invoke<MemoryStats>('get_memory_stats');
  } catch {
    return null;
  }
}
```
