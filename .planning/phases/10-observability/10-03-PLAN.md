import { invoke } from '@tauri-apps/api/core';

// EventKind discriminated union -- every logged event type
export type EventKind =
  | { kind: 'app_start'; version: string }
  | { kind: 'model_load'; model_id: string; duration_ms: number }
  | { kind: 'session_end'; metrics: import('./inference-monitor').SessionMetrics }
  | { kind: 'restart'; event: import('./session-manager').RestartEvent }
  | { kind: 'benchmark'; model_id: string; tok_per_sec: number; acceleration: string }
  | { kind: 'persona_check'; bloodline: string; score: number; passed: boolean; probe_id: string }
  | { kind: 'hw_detected'; vendor: string; vram_gb: number; acceleration: string; driver_ok: boolean }
  | { kind: 'error'; message: string; context: string };

export interface LogEntry {
  ts: string;           // ISO 8601
  session_id: string;   // UUID, stable per app launch
  event: EventKind;
}

let sessionId = crypto.randomUUID();
let buffer: LogEntry[] = [];
let recentEntries: LogEntry[] = [];  // In-memory last 100 for UI

/**
 * Append an event to the in-memory buffer and the on-disk events.jsonl log.
 * Writes are fire-and-forget via Tauri invoke.
 */
export function logEvent(event: EventKind): void {
  const entry: LogEntry = {
    ts: new Date().toISOString(),
    session_id: sessionId,
    event,
  };
  recentEntries = [entry, ...recentEntries].slice(0, 100);
  buffer.push(entry);
  // Flush: write all buffered entries, clear buffer
  flushBuffer();
}

async function flushBuffer(): Promise<void> {
  if (buffer.length === 0) return;
  const toWrite = [...buffer];
  buffer = [];
  try {
    // Tauri invoke to append lines to app_data/kr8tiv/events.jsonl
    // The command 'append_event_log' accepts a JSON string array of lines
    const lines = toWrite.map(e => JSON.stringify(e));
    await invoke('append_event_log', { lines });
  } catch (e) {
    // Don't crash the app if logging fails -- put back in buffer
    buffer = [...toWrite, ...buffer];
    console.warn('Event log flush failed:', e);
  }
}

export function getRecentEvents(limit = 50): LogEntry[] {
  return recentEntries.slice(0, limit);
}

export function getSessionId(): string {
  return sessionId;
}

use std::io::Write;
use tauri::Manager;

/// Append newline-delimited JSON lines to {app_data}/kr8tiv/events.jsonl.
/// Creates the file and directory if they don't exist.
#[tauri::command]
pub async fn append_event_log(app: tauri::AppHandle, lines: Vec<String>) -> Result<(), String> {
    let data_dir = app.path().app_data_dir()
        .map_err(|e| e.to_string())?;
    let kr8tiv_dir = data_dir.join("kr8tiv");
    fs::create_dir_all(&kr8tiv_dir).map_err(|e| e.to_string())?;
    let log_path = kr8tiv_dir.join("events.jsonl");
    let mut file = OpenOptions::new()
        .create(true)
        .append(true)
        .open(&log_path)
        .map_err(|e| e.to_string())?;
    for line in &lines {
        writeln!(file, "{}", line).map_err(|e| e.to_string())?;
    }
    Ok(())
}

export function wireAutoLogging(): void {
  // Import lazily to avoid circular deps
  import('./inference-monitor').then(({ subscribe }) => {
    let prevSessionId: string | null = null;
    subscribe(state => {
      // Log session end when a session transitions from active to complete
      if (state.sessionHistory.length > 0) {
        const latest = state.sessionHistory[0];
        if (latest.sessionId !== prevSessionId && latest.endedAt !== null) {
          prevSessionId = latest.sessionId;
          logEvent({ kind: 'session_end', metrics: latest });
        }
      }
    });
  });
  import('./session-manager').then(({ onRestart }) => {
    onRestart(event => {
      logEvent({ kind: 'restart', event });
    });
  });
}

import { logEvent } from './event-log';

export interface PersonaScore {
  score: number;          // 0.0 to 1.0 (1.0 = perfect persona fidelity)
  passed: boolean;        // true if score >= PASS_THRESHOLD
  probe_id: string;
  response_snippet: string;  // First 200 chars of response (for UI display)
  signals_found: string[];    // Which positive signals were detected
  signals_missing: string[];  // Which positive signals were expected but absent
  checked_at: string;         // ISO 8601
}

// Probe suite for Cipher (creative code companion)
// These prompts should reliably elicit Cipher's characteristic voice
const CIPHER_PROBES = [
  {
    id: 'cipher-creative-stack',
    prompt: 'What tech would you use for a visually stunning product landing page?',
    required_signals: ['three', 'gsap', 'lenis', 'webgl', 'shader', 'canvas', 'animation'],
    forbidden_signals: ['bootstrap', 'jquery', 'wordpress', 'template', 'simple'],
    min_required: 2,
  },
  {
    id: 'cipher-anti-slop',
    prompt: 'Show me a hero section component.',
    // Cipher should refuse or elevate generic requests -- NOT produce generic Tailwind hero
    required_signals: ['creative', 'visual', 'motion', 'craft', 'three', 'gsap', 'webgl'],
    forbidden_signals: ['hero-section', 'bg-gradient', 'animate-bounce', 'from-purple', 'welcome to'],
    min_required: 1,
  },
] as const;

const PASS_THRESHOLD = 0.6;      // Score >= 0.6 = persona healthy
const WARN_THRESHOLD = 0.4;      // Score 0.4-0.6 = persona drifting (amber)
const CHECK_INTERVAL_MESSAGES = 50;  // Run check every 50 messages

let messageCountSinceLastCheck = 0;
let lastScore: PersonaScore | null = null;
let listeners: Array<(score: PersonaScore) => void> = [];

export function onPersonaScore(fn: (score: PersonaScore) => void): () => void {
  listeners.push(fn);
  return () => { listeners = listeners.filter(l => l !== fn); };
}

export function getLastScore(): PersonaScore | null {
  return lastScore;
}

export function tickMessage(): void {
  messageCountSinceLastCheck++;
  if (messageCountSinceLastCheck >= CHECK_INTERVAL_MESSAGES) {
    messageCountSinceLastCheck = 0;
    // Fire-and-forget: run check asynchronously
    runPersonaCheck('cipher', 'cipher-kraken:latest').catch(console.error);
  }
}

/**
 * Run a persona health check for the given bloodline.
 * Sends canned probe prompts, scores responses, logs result.
 */
export async function runPersonaCheck(bloodline: string, modelId: string): Promise<PersonaScore> {
  // Use first probe for now; cycle through probes in future
  const probe = CIPHER_PROBES[messageCountSinceLastCheck % CIPHER_PROBES.length];

  let responseText = '';
  try {
    const res = await fetch('http://localhost:11434/api/chat', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        model: modelId,
        messages: [{ role: 'user', content: probe.prompt }],
        stream: false,          // Non-streaming for eval -- we need the full response
        options: { temperature: 0.3, num_predict: 150 },  // Low temp for consistency
      }),
    });
    const data = await res.json();
    responseText = data?.message?.content ?? '';
  } catch {
    responseText = '';
  }

  const lower = responseText.toLowerCase();
  const signalsFound = probe.required_signals.filter(sig => lower.includes(sig));
  const signalsMissing = probe.required_signals.filter(sig => !lower.includes(sig));
  const forbiddenHits = probe.forbidden_signals.filter(sig => lower.includes(sig));

  // Scoring:
  // +0.1 per required signal found (up to min_required checks)
  // -0.2 per forbidden signal hit
  // Base score from coverage: signalsFound.length / min_required, capped at 1.0
  const coverageScore = Math.min(signalsFound.length / probe.min_required, 1.0);
  const penaltyScore = Math.min(forbiddenHits.length * 0.2, 0.6);
  const rawScore = Math.max(0, coverageScore - penaltyScore);
  const score = Math.round(rawScore * 100) / 100;

  const result: PersonaScore = {
    score,
    passed: score >= PASS_THRESHOLD,
    probe_id: probe.id,
    response_snippet: responseText.slice(0, 200),
    signals_found: signalsFound,
    signals_missing: signalsMissing,
    checked_at: new Date().toISOString(),
  };

  lastScore = result;
  logEvent({ kind: 'persona_check', bloodline, score, passed: result.passed, probe_id: probe.id });
  listeners.forEach(fn => fn(result));
  return result;
}

"""
eval_pipeline.py -- KIN Model Evaluation Suite

Usage:
  python scripts/eval_pipeline.py --model cipher-kraken:latest --output results/eval.json
  python scripts/eval_pipeline.py --model cipher-kraken:latest --wandb --project kr8tiv-evals

Evaluates:

import argparse
import json
import time
import datetime
import requests
from typing import Optional

OLLAMA_BASE = "http://localhost:11434"
PASS_THRESHOLD = 0.6
WARN_THRESHOLD = 0.4

# === Persona Probes (mirrors TypeScript CIPHER_PROBES) ===
PERSONA_PROBES = [

GENERAL_PROBES = [

def call_ollama(model: str, prompt: str, system: str = "", temperature: float = 0.2) -> str:
    """Call Ollama /api/chat with a single user prompt. Returns response text."""
    messages = []
    if system:
        messages.append({"role": "system", "content": system})
    messages.append({"role": "user", "content": prompt})
    try:
        resp = requests.post(
            f"{OLLAMA_BASE}/api/chat",
            json={"model": model, "messages": messages, "stream": False,
                  "options": {"temperature": temperature, "num_predict": 200}},
            timeout=30,
        )
        resp.raise_for_status()
        return resp.json().get("message", {}).get("content", "")
    except Exception as e:
        return f"ERROR: {e}"


def score_persona_probe(probe: dict, response: str) -> dict:
    lower = response.lower()
    found = [s for s in probe["required_signals"] if s in lower]
    missing = [s for s in probe["required_signals"] if s not in lower]
    forbidden_hits = [s for s in probe["forbidden_signals"] if s in lower]
    coverage = min(len(found) / max(probe["min_required"], 1), 1.0)
    penalty = min(len(forbidden_hits) * 0.2, 0.6)
    score = max(0.0, coverage - penalty)
    return {
        "probe_id": probe["id"],
        "score": round(score, 3),
        "passed": score >= PASS_THRESHOLD,
        "signals_found": found,
        "signals_missing": missing,
        "forbidden_hits": forbidden_hits,
        "response_snippet": response[:300],
        "weight": probe["weight"],
    }


def run_eval_suite(model: str) -> dict:
    """Run full eval suite. Returns structured results dict."""
    results = {
        "model": model,
        "timestamp": datetime.datetime.utcnow().isoformat() + "Z",
        "persona_probes": [],
        "general_probes": [],
        "persona_score": 0.0,
        "general_score": 0.0,
        "overall_score": 0.0,
        "passed": False,
    }

    print(f"\n=== Persona Probes ({len(PERSONA_PROBES)}) ===")
    total_weight = 0.0
    weighted_sum = 0.0
    for probe in PERSONA_PROBES:
        print(f"  Running: {probe['id']}...", end=" ", flush=True)
        response = call_ollama(model, probe["prompt"], probe.get("system", ""), temperature=0.3)
        result = score_persona_probe(probe, response)
        results["persona_probes"].append(result)
        weighted_sum += result["score"] * probe["weight"]
        total_weight += probe["weight"]
        status = "PASS" if result["passed"] else "FAIL"
        print(f"{status} (score={result['score']:.2f}, found={result['signals_found']})")
    results["persona_score"] = round(weighted_sum / total_weight, 3) if total_weight > 0 else 0.0

    print(f"\n=== General Capability Probes ({len(GENERAL_PROBES)}) ===")
    total_weight = 0.0
    weighted_sum = 0.0
    for probe in GENERAL_PROBES:
        print(f"  Running: {probe['id']}...", end=" ", flush=True)
        response = call_ollama(model, probe["prompt"], temperature=0.1)
        passed = bool(probe["check"](response))
        result = {"probe_id": probe["id"], "passed": passed, "weight": probe["weight"],
                  "response_snippet": response[:300]}
        results["general_probes"].append(result)
        weighted_sum += (1.0 if passed else 0.0) * probe["weight"]
        total_weight += probe["weight"]
        print(f"{'PASS' if passed else 'FAIL'}")
    results["general_score"] = round(weighted_sum / total_weight, 3) if total_weight > 0 else 0.0

    # Overall: 60% persona, 40% general (persona is the core differentiator)
    results["overall_score"] = round(
        results["persona_score"] * 0.6 + results["general_score"] * 0.4, 3
    )
    # Reject any checkpoint where general capabilities drop > 5% below 1.0 (per PITFALLS.md Pitfall 4)
    results["passed"] = (
        results["persona_score"] >= PASS_THRESHOLD and
        results["general_score"] >= 0.75  # Must maintain >= 75% general capability
    )
    return results


def log_to_wandb(results: dict, project: str, run_name: Optional[str] = None) -> None:
    """Log eval results to W&B. Requires wandb installed: pip install wandb."""
    try:
        import wandb
        run = wandb.init(project=project, name=run_name or f"eval-{results['model']}", config={
            "model": results["model"],
            "eval_timestamp": results["timestamp"],
        })
        # Log aggregate metrics
        wandb.log({
            "eval/persona_score": results["persona_score"],
            "eval/general_score": results["general_score"],
            "eval/overall_score": results["overall_score"],
            "eval/passed": int(results["passed"]),
        })
        # Log per-probe scores
        for probe_result in results["persona_probes"]:
            wandb.log({f"persona/{probe_result['probe_id']}": probe_result["score"]})
        for probe_result in results["general_probes"]:
            wandb.log({f"general/{probe_result['probe_id']}": int(probe_result["passed"])})
        run.finish()
        print(f"\nW&B run logged: {run.url}")
    except ImportError:
        print("\nW&B not installed. Run: pip install wandb")
    except Exception as e:
        print(f"\nW&B logging failed: {e}")
