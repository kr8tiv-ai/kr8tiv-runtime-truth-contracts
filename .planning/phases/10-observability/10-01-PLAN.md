```rust
use serde::{Deserialize, Serialize};
use std::process::Command;

#[derive(Debug, Serialize, Deserialize, Clone)]
pub struct GpuInfo {
    pub vendor: String,           // "nvidia" | "amd" | "apple" | "unknown"
    pub vram_gb: f32,             // 0.0 if unknown or CPU-only
    pub acceleration_mode: String, // "cuda" | "metal" | "rocm" | "cpu"
    pub gpu_name: String,         // e.g. "NVIDIA GeForce RTX 4070"
    pub driver_ok: bool,          // true if driver version is sufficient
}

/// Detect GPU info via platform-specific commands and Ollama /api/ps.
/// Priority: Ollama ps (authoritative) > nvidia-smi > system info > fallback.
#[tauri::command]
pub async fn get_hw_info() -> Result<GpuInfo, String> {
    // Step 1: Try nvidia-smi for NVIDIA GPU (Windows/Linux)
    if let Ok(output) = Command::new("nvidia-smi")
        .args(["--query-gpu=name,memory.total,driver_version", "--format=csv,noheader,nounits"])
        .output()
    {
        if output.status.success() {
            let text = String::from_utf8_lossy(&output.stdout);
            let parts: Vec<&str> = text.trim().splitn(3, ',').collect();
            if parts.len() >= 2 {
                let name = parts[0].trim().to_string();
                let vram_mb: f32 = parts[1].trim().parse().unwrap_or(0.0);
                let driver = if parts.len() > 2 { parts[2].trim() } else { "" };
                // CUDA requires driver >= 531 for modern Ollama
                let driver_ver: f32 = driver.split('.').next()
                    .and_then(|s| s.parse().ok()).unwrap_or(0.0);
                return Ok(GpuInfo {
                    vendor: "nvidia".to_string(),
                    vram_gb: vram_mb / 1024.0,
                    acceleration_mode: "cuda".to_string(),
                    gpu_name: name,
                    driver_ok: driver_ver >= 531.0,
                });
            }
        }
    }

    // Step 2: Try system_profiler on macOS for Metal
    #[cfg(target_os = "macos")]
    if let Ok(output) = Command::new("system_profiler")
        .args(["SPDisplaysDataType", "-json"])
        .output()
    {
        if output.status.success() {
            let text = String::from_utf8_lossy(&output.stdout);
            // Parse VRAM from JSON -- look for "VRAM (Total)"
            if let Some(start) = text.find("VRAM") {
                let snippet = &text[start..start.min(text.len()).min(start + 100)];
                let vram_gb: f32 = snippet.chars()
                    .filter(|c| c.is_ascii_digit() || *c == '.')
                    .collect::<String>()
                    .parse()
                    .unwrap_or(0.0);
                return Ok(GpuInfo {
                    vendor: "apple".to_string(),
                    vram_gb,
                    acceleration_mode: "metal".to_string(),
                    gpu_name: "Apple Silicon GPU".to_string(),
                    driver_ok: true, // Metal always available on Apple Silicon
                });
            }
        }
    }

    // Step 3: Fallback -- CPU-only
    Ok(GpuInfo {
        vendor: "unknown".to_string(),
        vram_gb: 0.0,
        acceleration_mode: "cpu".to_string(),
        gpu_name: "CPU Only".to_string(),
        driver_ok: false,
    })
}
```

```typescript
import { invoke } from '@tauri-apps/api/core';

export interface GpuInfo {
  vendor: 'nvidia' | 'amd' | 'apple' | 'unknown';
  vram_gb: number;
  acceleration_mode: 'cuda' | 'metal' | 'rocm' | 'cpu';
  gpu_name: string;
  driver_ok: boolean;
}

export interface SessionMetrics {
  sessionId: string;
  startedAt: number;          // Date.now()
  endedAt: number | null;
  totalTokens: number;
  durationMs: number;
  avgTokPerSec: number;
  minTokPerSec: number;
  maxTokPerSec: number;
}

export interface MonitorState {
  gpuInfo: GpuInfo | null;
  currentTokPerSec: number;
  sessionHistory: SessionMetrics[];   // Last 20 sessions
  benchmarkTokPerSec: number | null;  // First-run speed test result
  isLow: boolean;                     // true when below LOW_TOK_THRESHOLD
}

const LOW_TOK_THRESHOLD = 10;         // tok/s below this = warn (per PITFALLS.md)
const EXPECTED_GPU_MINIMUM = 15;      // GPU should manage at least this

let state: MonitorState = {
  gpuInfo: null,
  currentTokPerSec: 0,
  sessionHistory: [],
  benchmarkTokPerSec: null,
  isLow: false,
};

let listeners: Array<(s: MonitorState) => void> = [];
let currentSession: { id: string; start: number; tokSamples: number[] } | null = null;

function notify() {
  listeners.forEach(fn => fn({ ...state, sessionHistory: [...state.sessionHistory] }));
}

export function subscribe(fn: (s: MonitorState) => void): () => void {
  listeners.push(fn);
  fn({ ...state });
  return () => { listeners = listeners.filter(l => l !== fn); };
}

export async function initialize(): Promise<GpuInfo> {
  try {
    state.gpuInfo = await invoke<GpuInfo>('get_hw_info');
  } catch {
    state.gpuInfo = { vendor: 'unknown', vram_gb: 0, acceleration_mode: 'cpu', gpu_name: 'Unknown', driver_ok: false };
  }
  notify();
  return state.gpuInfo;
}

export function startSession(sessionId: string): void {
  currentSession = { id: sessionId, start: Date.now(), tokSamples: [] };
}

export function recordChunk(tokPerSec: number): void {
  if (tokPerSec <= 0) return;
  state.currentTokPerSec = tokPerSec;

  // Warning: running on CPU-capable hardware but tok/s is very low
  state.isLow = tokPerSec < LOW_TOK_THRESHOLD &&
    (state.gpuInfo?.acceleration_mode !== 'cpu' || tokPerSec < 3);

  if (currentSession) {
    currentSession.tokSamples.push(tokPerSec);
  }
  notify();
}

export function endSession(): SessionMetrics | null {
  if (!currentSession || currentSession.tokSamples.length === 0) return null;
  const samples = currentSession.tokSamples;
  const durationMs = Date.now() - currentSession.start;
  const metrics: SessionMetrics = {
    sessionId: currentSession.id,
    startedAt: currentSession.start,
    endedAt: Date.now(),
    totalTokens: samples.length,
    durationMs,
    avgTokPerSec: samples.reduce((a, b) => a + b, 0) / samples.length,
    minTokPerSec: Math.min(...samples),
    maxTokPerSec: Math.max(...samples),
  };
  state.sessionHistory = [metrics, ...state.sessionHistory].slice(0, 20);
  currentSession = null;
  state.currentTokPerSec = 0;
  notify();
  return metrics;
}

/**
 * Run a speed benchmark: send a fixed 50-token prompt, measure tok/s.
 * Called once after first model load. Stores result in state.benchmarkTokPerSec.
 * Uses the existing streamChat from ollama service indirectly via Ollama /api/generate.
 */
export async function runSpeedBenchmark(modelId: string): Promise<number> {
  const BENCHMARK_PROMPT = 'Count from 1 to 50 separated by spaces. Do not add any other text.';
  const start = Date.now();
  let tokenCount = 0;
  try {
    const response = await fetch('http://localhost:11434/api/generate', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ model: modelId, prompt: BENCHMARK_PROMPT, stream: true }),
    });
    const reader = response.body!.getReader();
    const decoder = new TextDecoder();
    while (true) {
      const { done, value } = await reader.read();
      if (done) break;
      const lines = decoder.decode(value).split('\n').filter(Boolean);
      for (const line of lines) {
        try {
          const chunk = JSON.parse(line);
          if (!chunk.done) tokenCount++;
          if (chunk.done && chunk.eval_count && chunk.eval_duration) {
            const tokPerSec = (chunk.eval_count / chunk.eval_duration) * 1e9;
            state.benchmarkTokPerSec = tokPerSec;
            state.isLow = tokPerSec < LOW_TOK_THRESHOLD;
            notify();
            return tokPerSec;
          }
        } catch { /* skip malformed */ }
      }
    }
  } catch (e) {
    console.error('Benchmark failed:', e);
  }
  // Fallback: estimate from elapsed time
  const elapsed = (Date.now() - start) / 1000;
  const estimate = elapsed > 0 ? tokenCount / elapsed : 0;
  state.benchmarkTokPerSec = estimate;
  notify();
  return estimate;
}

export function getMonitorState(): MonitorState {
  return { ...state, sessionHistory: [...state.sessionHistory] };
}
```
